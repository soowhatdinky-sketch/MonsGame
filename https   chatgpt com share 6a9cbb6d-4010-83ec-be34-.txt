builder = '''extends Node3D
@export var kit_path := "res://Creepy_Cat/Sci-Fi_Starter_Kit/Prefabs"
@export var prefab_limit := 72

func _ready()->void:
    call_deferred("_populate")

func _populate()->void:
    var scenes:=_find_scenes(kit_path)
    var usable:Array[String]=[]
    for path in scenes:
        var lower:=path.to_lower()
        if "wall_" in lower or "floor_" in lower or "door_" in lower or "beam_" in lower or "pipe" in lower:
            usable.append(path)
    for i in range(mini(prefab_limit,usable.size())):
        var packed=load(usable[i])
        if packed is PackedScene:
            var item=packed.instantiate()
            item.position=_slot(i)
            add_child(item)

func _find_scenes(path:String)->Array[String]:
    var result:Array[String]=[]
    var dir:=DirAccess.open(path)
    if dir==null:return result
    dir.list_dir_begin()
    var item:=dir.get_next()
    while item!="":
        if item!="." and item!="..":
            var full:=path.path_join(item)
            if dir.current_is_dir(): result.append_array(_find_scenes(full))
            elif item.to_lower().ends_with(".tscn"): result.append(full)
        item=dir.get_next()
    dir.list_dir_end()
    return result

func _slot(i:int)->Vector3:
    var cols:=6
    var col:=i%cols
    var row:=(i/cols)%2
    var deck:=i/(cols*2)
    return Vector3((col-2.5)*95.0,(deck%12)*32.0-150.0,(row-0.5)*460.0+(deck/12)*520.0-520.0)
'''
