import requests
import time
import os
import hashlib

def download_with_resume(url, filename):
    """Downloads a file and resumes if the connection drops."""
    headers = {}
    if os.path.exists(filename):
        headers['Range'] = f"bytes={os.path.getsize(filename)}-"
    
    print(f"--- Starting/Resuming download of {filename} ---")
    
    while True:
        try:
            # stream=True is required for large files to avoid crashing RAM
            response = requests.get(url, headers=headers, stream=True, timeout=15)
            mode = 'ab' if 'Range' in headers else 'wb'
            
            with open(filename, mode) as f:
                for chunk in response.iter_content(chunk_size=1024 * 1024): # 1MB chunks
                    if chunk:
                        f.write(chunk)
            
            print("\nDownload finished successfully.")
            break
        except (requests.exceptions.RequestException, ConnectionError):
            print("Network dropped! Waiting 10 seconds to retry...")
            time.sleep(10)
            headers['Range'] = f"bytes={os.path.getsize(filename)}-"
